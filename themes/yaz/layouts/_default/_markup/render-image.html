{{- /* Run markdown images through Hugo's image pipeline: capped at 1200px wide and converted to WebP, so the large @2x source PNGs are never served raw. Animated GIFs, SVGs and remote images pass through untouched. */ -}} {{- $dest := .Destination -}} {{- $ext := lower (path.Ext $dest) -}} {{- $passthrough := in (slice ".gif" ".svg") $ext -}} {{- $img := false -}} {{- if not $passthrough -}} {{- $img = resources.Get (strings.TrimPrefix "/" $dest) -}} {{- end -}} {{- $src := $dest -}} {{- $w := 0 -}} {{- $h := 0 -}} {{- if $img -}} {{- /* Cap at 2400 so native @2x screenshots aren't downscaled; never upscale. */ -}} {{- $target := cond (gt $img.Width 2400) 2400 $img.Width -}} {{- $opt := $img.Resize (printf "%dx webp q92" $target) -}} {{- $src = $opt.RelPermalink -}} {{- $w = $opt.Width -}} {{- $h = $opt.Height -}} {{- end -}}
